#include <Arduino.h>
#include <Wire.h>
#include <WiFi.h>
#include <WebServer.h>
#include <WebSocketsServer.h>
#include <ArduinoJson.h>
#include <math.h>

// --- ADS1115 ---
#include <Adafruit_ADS1X15.h>
Adafruit_ADS1X15 ads;

// --- BME680 ---
#include <Adafruit_Sensor.h>
#include <Adafruit_BME680.h>
Adafruit_BME680 bme;

// --- GPS ---
#include <SparkFun_u-blox_GNSS_v3.h>
SFE_UBLOX_GNSS gps;

// --- BMA400 ---
#include "SparkFun_BMA400_Arduino_Library.h"
BMA400 accelerometer;
uint8_t i2cAddress = 0x15;

// --- Capteur résistif ADS1115 ---
const float VCC = 3.3;
const float R1 = 100000.0;

// --- MAX30101 BPM ---
#include "MAX30105.h"
#include "heartRate.h"
MAX30105 particleSensor;

const byte RATE_SIZE = 4;
byte rates[RATE_SIZE];
byte rateSpot = 0;
long lastBeat = 0;
float beatsPerMinute = 0;
volatile int beatAvg = 0;
#define IR_SEUIL_DOIGT 80000

// --- Transpiration baseline ---
float baselineR2 = -1.0;
unsigned long startTime = 0;

// --- Dernières valeurs valides ---
float lastValidTranspi = 0.0;
float lastValidTemperature = 0.0;
int lastValidBpm = 0;
float lastValidImuX = 0.0, lastValidImuY = 0.0, lastValidImuZ = 0.0;

// --- Podomètre (tâche dédiée 50Hz) ---
volatile uint32_t stepCount = 0;
bool peakDetected = false;
unsigned long lastStepTime = 0;
const float STEP_THRESHOLD = 1.2;
const unsigned long STEP_MIN_MS = 250;
TaskHandle_t stepTask;

// --- WiFi AP ---
const char* AP_SSID = "ESP32-Capteurs";
const char* AP_PASS = "12345678";

// --- Serveurs ---
WebServer httpServer(80);
WebSocketsServer wsServer(81);

// --- Timers ---
unsigned long lastBroadcastTime = 0;
const unsigned long BROADCAST_INTERVAL = 500;

// --- Task BPM ---
TaskHandle_t bpmTask;

// ======================================================
//                   TASK BPM (Core 0)
// ======================================================
void bpmLoop(void* pvParameters) {
  for (;;) {
    long irValue = particleSensor.getIR();

    if (irValue >= IR_SEUIL_DOIGT) {
      if (checkForBeat(irValue)) {
        long delta = millis() - lastBeat;
        lastBeat = millis();
        if (delta >= 200 && delta <= 3000) {
          beatsPerMinute = 60 / (delta / 1000.0);
          if (beatsPerMinute > 40 && beatsPerMinute < 255) {
            rates[rateSpot++] = (byte)beatsPerMinute;
            rateSpot %= RATE_SIZE;
            int sum = 0, count = 0;
            for (byte i = 0; i < RATE_SIZE; i++) {
              if (rates[i] >= 40) { sum += rates[i]; count++; }
            }
            beatAvg = (count > 0) ? (sum / count) : 0;
          }
        }
      }
    } else {
      rateSpot = 0; beatAvg = 0; lastBeat = 0; beatsPerMinute = 0;
      for (byte i = 0; i < RATE_SIZE; i++) rates[i] = 0;
    }
    vTaskDelay(1);
  }
}

// ======================================================
//                   TASK PODOMÈTRE (Core 0, 50Hz)
// ======================================================
void stepLoop(void* pvParameters) {
  for (;;) {
    accelerometer.getSensorData();

    float x = accelerometer.data.accelX;
    float y = accelerometer.data.accelY;
    float z = accelerometer.data.accelZ;
    float magnitude = sqrt(x*x + y*y + z*z);

    unsigned long now = millis();

    if (magnitude > STEP_THRESHOLD && !peakDetected) {
      if (now - lastStepTime > STEP_MIN_MS) {
        stepCount++;
        lastStepTime = now;
      }
      peakDetected = true;
    } else if (magnitude <= STEP_THRESHOLD) {
      peakDetected = false;
    }

    // Mettre à jour les valeurs IMU partagées
    if (x != 0 || y != 0 || z != 0) {
      lastValidImuX = x;
      lastValidImuY = y;
      lastValidImuZ = z;
    }

    vTaskDelay(20 / portTICK_PERIOD_MS); // 50Hz
  }
}

// ======================================================
//                   BUILD JSON PAYLOAD
// ======================================================
String buildPayload(int bpmValue) {
  JsonDocument doc;

  // BPM
  if (bpmValue > 0) lastValidBpm = bpmValue;
  doc["bpm"] = lastValidBpm;

  // IMU
  JsonObject imu = doc["imu"].to<JsonObject>();
  imu["x"] = lastValidImuX;
  imu["y"] = lastValidImuY;
  imu["z"] = lastValidImuZ;

  // Steps
  doc["steps"] = stepCount;

  // Transpiration
  int16_t adcValue = ads.readADC_SingleEnded(0);
  float Vout = adcValue * (VCC / 32767.0);
  if (Vout > 0.1 && Vout < (VCC - 0.1)) {
    float R2 = R1 * Vout / (VCC - Vout);
    if (baselineR2 < 0 && millis() - startTime > 5000) {
      baselineR2 = R2;
      Serial.print("Baseline transpi: ");
      Serial.print(baselineR2 / 1000.0);
      Serial.println(" kΩ");
    }
    if (baselineR2 > 0) {
      float variation = (baselineR2 - R2) / baselineR2 * 100.0;
      float transpiPercent = constrain(variation * 10.0, 0.0, 100.0);
      if (transpiPercent > 0) lastValidTranspi = transpiPercent;
    }
  }
  doc["transpiration"] = round(lastValidTranspi);

  // Température BME680
  if (bme.performReading()) {
    lastValidTemperature = bme.temperature;
  }
  doc["temperature"] = lastValidTemperature;

  // GPS
  JsonObject gpsObj = doc["gps"].to<JsonObject>();
  gpsObj["lat"] = gps.getLatitude() / 1e7;
  gpsObj["lng"] = gps.getLongitude() / 1e7;
  gpsObj["alt"] = gps.getAltitude() / 1000.0;

  String output;
  serializeJson(doc, output);
  return output;
}

// ======================================================
//                      SETUP
// ======================================================
void setup() {
  Serial.begin(115200);
  delay(2000);
  Serial.println("\n=== Initialisation des capteurs ===");

  Wire.begin();
  delay(200);

  // WiFi AP
  WiFi.mode(WIFI_AP);
  delay(100);
  // Forcer l'IP statique 192.168.4.1
  IPAddress local_ip(192, 168, 4, 1);
  IPAddress gateway(192, 168, 4, 1);
  IPAddress subnet(255, 255, 255, 0);
  WiFi.softAPConfig(local_ip, gateway, subnet);
  WiFi.softAP(AP_SSID, AP_PASS, 6);
  delay(500);
  Serial.print("AP démarré - IP: ");
  Serial.println(WiFi.softAPIP());

  // HTTP
  httpServer.on("/data", []() {
    httpServer.sendHeader("Access-Control-Allow-Origin", "*");
    httpServer.send(200, "application/json", buildPayload(beatAvg));
  });
  httpServer.begin();
  Serial.println("HTTP port 80 -> /data");

  // WebSocket
  wsServer.begin();
  Serial.println("WebSocket port 81");

  // ADS1115
  if (!ads.begin()) {
    Serial.println("ADS1115 non detecte !");
    while (1);
  }
  Serial.println("ADS1115 detecte");
  startTime = millis();

  // BME680
  if (!bme.begin(0x76)) {
    if (!bme.begin(0x77)) {
      Serial.println("BME680 non detecte !");
    } else {
      Serial.println("BME680 detecte (0x77)");
    }
  } else {
    Serial.println("BME680 detecte (0x76)");
  }
  bme.setTemperatureOversampling(BME680_OS_8X);
  bme.setHumidityOversampling(BME680_OS_2X);
  bme.setPressureOversampling(BME680_OS_4X);
  bme.setIIRFilterSize(BME680_FILTER_SIZE_3);

  // GPS
  if (!gps.begin()) {
    Serial.println("GPS non detecte ! (ignoré)");
  } else {
    Serial.println("GPS detecte");
  }

  // BMA400
  while (accelerometer.beginI2C(i2cAddress) != BMA400_OK) {
    Serial.println("BMA400 non detecte !");
    delay(1000);
  }
  Serial.println("BMA400 detecte");

  // MAX30101
  if (!particleSensor.begin(Wire, I2C_SPEED_STANDARD)) {
    Serial.println(" MAX30101 non detecte !");
    while (1);
  }
  particleSensor.setup();
  particleSensor.setPulseAmplitudeRed(0x1F);
  particleSensor.setPulseAmplitudeGreen(0);
  Serial.println(" MAX30101 detecte");

  // Task BPM sur Core 0
  xTaskCreatePinnedToCore(bpmLoop, "BPM", 2048, NULL, 1, &bpmTask, 0);
  Serial.println("Task BPM démarrée sur Core 0");

  // Task podomètre sur Core 0
  xTaskCreatePinnedToCore(stepLoop, "STEP", 2048, NULL, 1, &stepTask, 0);
  Serial.println("Task podomètre démarrée sur Core 0");

  Serial.println("\n=== Démarrage ===\n");
}

// ======================================================
//                      LOOP (Core 1)
// ======================================================
void loop() {
  httpServer.handleClient();
  wsServer.loop();

  unsigned long now = millis();

  if (now - lastBroadcastTime >= BROADCAST_INTERVAL) {
    lastBroadcastTime = now;
    gps.checkUblox();
    String payload = buildPayload(beatAvg);
    wsServer.broadcastTXT(payload);
    Serial.println(payload);
  }
}
